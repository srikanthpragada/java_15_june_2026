);

CREATE TABLE EMPLOYEES (
    EMP_ID       INTEGER      PRIMARY KEY AUTOINCREMENT,
    EMP_NAME     VARCHAR (30) NOT NULL,
    EMP_JOB      VARCHAR (10) DEFAULT 'JP',
    EMP_SALARY   INTEGER      CHECK (EMP_SALARY > 100000),
    EMP_DEPT_ID  VARCHAR (2)  REFERENCES DEPARTMENTS(DEPT_ID)
);
